# Root types for mq-emu-bindgen. Each is generated together with every record
# it contains by value; types only reached through pointers become opaque.
# After editing, run `cargo run -p mq-emu-bindgen` and commit the result.

PlayerClient            # a spawn: you, other players, NPCs, corpses (pLocalPlayer, pTarget)
PcClient                # your character's data (pLocalPC)
PcZoneClient            # PcClient's base whose methods take it as `this` (DoCombatAbility)
CharacterBase           # PcClient's base whose methods take it as `this` (GetItemByGlobalIndex)
PlayerZoneClient        # PlayerClient's base, as the game's methods take a spawn (UseSkill)
PcProfile               # your character's current profile: inventory, buffs, spell book, gems
ProfileList             # PcClient::ProfileManager's lists of profiles
PlayerManagerClient     # the spawn list (pSpawnManager)
KeypressHandler         # movement key state (pKeypressHandler)
zoneHeader              # the zone you're in (pZoneInfo)
EQWorldData             # every zone the client knows, by ID (pWorldData)
EQZoneInfo              # one entry of EQWorldData::ZoneArray
EQGroundItem            # an item on the ground (OnAddGroundItem)
EQGroundItemListManager # the ground item list (pItemList)
EverQuestinfo           # client-wide state, incl. which game commands are down (pEverQuestInfo)
ItemClient              # an item instance: in inventory, a bag, the bank, on the ground
VeBaseReferenceCount    # ItemClient's reference count, which a VePointer<ItemClient> holds
ItemDefinition          # what an item is (name, stats, effects), shared by every instance
ClientSpellManager      # the spell table (pSpellMgr)
EQ_Spell                # one spell
CGroup                  # your group (PcClient::Group)
CGroupMember            # one group member
CStrRep                 # the text behind a CXStr
CTargetWnd              # the target window: your target's buffs as the game shows them (pTargetWnd)
CPetInfoWnd             # the pet window: your pet's buffs and commands (pPetInfoWnd)
CRaid                   # your raid (pRaid)
CMercenaryManager       # your mercenary (pMercManager)
ExtendedTargetList      # the extended target window's slots (PcClient::pExtendedTargetList)
AggroMeterManagerClient # aggro percentages (pAggroInfo)
AltAdvManager           # every alternate ability (pAltAdvManager)
CAltAbilityData         # one alternate ability rank
SkillManager            # skill definitions and caps (pSkillMgr)
EQ_Skill                # one skill
CMerchantWnd            # the merchant window (pMerchantWnd)
MerchantPageHandler     # one merchant window page: the items for sale, buyback
CLootWnd                # the loot window (pLootWnd)
ExtendedTargetSlot      # one extended target slot
HashTable<CXStr,int,ResizePolicyNoResize>::HashNode # a buff window's caster names, by spell ID
CXWnd                   # an EQ UI window or control (FindMQ2Window, its children)
CXWndManager            # every window, focus (pWndMgr)
ControllerBase          # a window's controller (CXWnd::pController), told of its notifications
CCastSpellWnd           # the spell gem bar (pCastSpellWnd): which gems are shown
CSpellGemWnd            # one spell gem (CCastSpellWnd::SpellSlots)
CBuffWindow             # the buff and song windows (pBuffWnd, pSongWnd): spell IDs and timers
CCastingWnd             # the casting window (pCastingWnd), shown while you cast
CSpellBookWnd           # the spell book (pSpellBookWnd), which memorizes spell sets
CButtonWnd              # a button or checkbox
CListWnd                # a list (and a combo box's drop-down, a context menu)
SListWndLine            # one row of a list
SListWndCell            # one cell of a list row
CComboWnd               # a combo box
CTabWnd                 # a tab box
CPageWnd                # one page of a tab box
CGaugeWnd               # a gauge (HP, mana, progress bars)
CStmlWnd                # rich text
CContextMenuManager     # the open context menus (pContextMenuManager)
CContextMenu            # one context menu
CInvSlotWnd             # an item slot window (inventory, bags, bank, loot, trade, hot buttons)
CInvSlotMgr             # every item slot window (pInvSlotMgr), as /itemnotify finds them
CInvSlot                # one item slot: its window
CCursorAttachment       # what the mouse carries (pCursorAttachment): an item, a spell...
CHotButton              # one hot button
CHotButtonWnd           # a hot bar (HotButtonWnd, HotButtonWnd2...)
CContainerMgr           # the open bag windows (pContainerMgr), as FindMQ2Window finds pack1, bank1, enviro
CContainerWnd           # one bag's window: its bag and where it is
CTradeWnd               # the trade window (pTradeWnd)
CItemDisplayManager     # the item display windows (pItemDisplayManager)
CSpellDisplayManager    # the spell display windows (pSpellDisplayManager)
CTaskManager            # your tasks: solo quests and the shared task (pTaskManager)
DatabaseStringTable     # the game's string table (pDBStr)
ClientAuraManager       # your auras (pAuraMgr), as the aura window lists them
AuraData                # one aura (ClientAuraManager::Auras)
StringTable             # the game's token strings (pStringTable): skill names
CEverQuest              # the game's top object (pEverQuest): class and race names
CMDLIST                 # one of the game's slash commands (EQADDR_CMDLIST): name and handler
SharedTaskPlayerInfo    # one member of your shared task (pTaskMember, then each pNext)
AchievementManager      # every achievement, its categories and your progress (AchievementManager::Instance)
AchievementCategory     # one achievement category
Achievement             # one achievement
SingleAchievementAndComponentsInfo # your progress on one achievement
AchievementComponent    # one objective of an achievement
AchievementInfo         # one achievement in a category
CTribute                # tribute state (pTribute)
UniversalChatProxy      # the chat service: your friends list (pChatService)
CDisplay                # the display: the game clock in ms (TimeStamp) (pDisplay)
CXMLData                # a window's XML definition: its name, screen ID and type name
CTextureAnimation       # an animated texture (its name)
CTextObjectInterface    # a text object, as an item slot shows its count
CSidlManager            # the UI's XML manager (pSidlMgr)
CSliderWnd              # a slider
CParamScreenPiece       # a screen piece's XML parameters: its screen ID (CXMLData::ScreenID)
